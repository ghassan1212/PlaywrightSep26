pipeline {
    agent any

    stages {
        stage('Install Dependencies') {
            steps {
                bat '''
                    cd /d C:\\Play_Test
                    npm install
                '''
            }
        }

        stage('Run Playwright Tests') {
            steps {
                catchError(buildResult: 'FAILURE', stageResult: 'FAILURE') {
                    bat '''
                        cd /d C:\\Play_Test
                        if exist allure-results rmdir /s /q allure-results
                        npx playwright test
                    '''
                }
            }
        }

        stage('Copy Allure Results') {
            steps {
                bat '''
                    if exist allure-results rmdir /s /q allure-results
                    xcopy /E /I /Y C:\\Play_Test\\allure-results allure-results
                '''
            }
        }
    }

    post {
        always {
            allure([
                [path: 'allure-results']
            ])
        }
    }
}
